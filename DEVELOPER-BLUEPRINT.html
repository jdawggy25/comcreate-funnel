📦 Data Layer Implementation

CRITICAL

The Data Layer MUST be initialized BEFORE the GTM container code loads.

● data-layer-init.js - Add BEFORE GTM script
// Initialize Data Layer - MUST BE BEFORE GTM
window.dataLayer = window.dataLayer || [];

// Helper: Get URL parameters
function getUrlParam(name) {
  return new URLSearchParams(window.location.search).get(name);
}

// Store GCLID in cookie for cross-session attribution
(function() {
  const gclid = getUrlParam('gclid');
  if (gclid) {
    const expiry = new Date();
    expiry.setDate(expiry.getDate() + 90);
    document.cookie = 'gclid=' + gclid + ';expires=' + expiry.toUTCString() + ';path=/;SameSite=Lax';
  }
})();

// Push initial page data
dataLayer.push({
  'event': 'page_data_ready',
  'page': {
    'type': 'landing_page',
    'service_category': 'plumbing', // Change per page
    'service_type': 'emergency',    // emergency, scheduled, maintenance
    'location': {
      'city': 'Austin',
      'state': 'TX'
    }
  },
  'campaign': {
    'source': getUrlParam('utm_source') || 'direct',
    'medium': getUrlParam('utm_medium') || 'none',
    'campaign': getUrlParam('utm_campaign') || 'none',
    'gclid': getUrlParam('gclid') || 'none'
  },
  'business': {
    'name': 'Your Business Name',
    'phone': '+15125551234'
  }
});

// Track page load time for engagement metrics
window.pageLoadTime = Date.now();

Lead Value Calculator

● lead-value.js
// Dynamic lead value based on service type and urgency
const LeadValues = {
  base: {
    'emergency_plumbing': 250,
    'drain_cleaning': 150,
    'water_heater': 400,
    'hvac_repair': 300,
    'ac_installation': 2000,
    'electrical': 200,
    'roofing': 800
  },
  urgencyMultiplier: {
    'emergency': 1.5,
    'same_day': 1.2,
    'this_week': 1.0,
    'flexible': 0.8
  }
};

function calculateLeadValue(service, urgency) {
  const base = LeadValues.base[service] || 100;
  const multiplier = LeadValues.urgencyMultiplier[urgency] || 1.0;
  return Math.round(base * multiplier);
}

// Track all phone link clicks
document.addEventListener('DOMContentLoaded', function() {
  document.querySelectorAll('a[href^="tel:"]').forEach(function(link) {
    link.addEventListener('click', function(e) {
      const phoneNumber = this.href.replace('tel:', '');
      const location = this.dataset.location || 'unknown';

      dataLayer.push({
        'event': 'phone_click',
        'phone_number': phoneNumber,
        'click_location': location,
        'click_text': this.innerText.trim(),
        'time_on_page': Math.round((Date.now() - window.pageLoadTime) / 1000)
      });
    });
  });
});

📝 Form Submission Tracking

● form-tracking.js - Complete Implementation
const FormTracker = {
  formStarted: false,

  init: function() {
    const form = document.querySelector('#lead-form');
    if (!form) return;

    this.trackFormStart(form);
    this.trackSubmission(form);
  },

  trackFormStart: function(form) {
    const self = this;
    form.querySelectorAll('input, select, textarea').forEach(field => {
      field.addEventListener('focus', function() {
        if (!self.formStarted) {
          self.formStarted = true;
          dataLayer.push({
            'event': 'form_start',
            'form_id': form.id,
            'first_field': this.name
          });
        }
      }, { once: false });
    });
  },

  trackSubmission: function(form) {
    form.addEventListener('submit', (e) => {
      e.preventDefault();

      const formData = new FormData(form);
      const data = Object.fromEntries(formData);

      // Calculate lead value
      const leadValue = calculateLeadValue(data.service, data.urgency);

      // Generate transaction ID for deduplication
      const transactionId = 'lead_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);

      // Push conversion event
      dataLayer.push({
        'event': 'form_submission',
        'form_id': form.id,
        'form_name': form.dataset.formName || 'Lead Form',
        'service_requested': data.service,
        'urgency': data.urgency,
        'lead_value': leadValue,
        'transaction_id': transactionId,

        // Enhanced Conversions data
        'user_data': {
          'email': data.email,
          'phone_number': this.normalizePhone(data.phone),
          'address': {
            'postal_code': data.zip,
            'country': 'US'
          }
        }
      });

      // Submit form via AJAX
      this.submitForm(form, data, transactionId);
    });
  },

  normalizePhone: function(phone) {
    if (!phone) return '';
    const cleaned = phone.replace(/\D/g, '');
    return cleaned.length === 10 ? '+1' + cleaned : '+' + cleaned;
  },

  submitForm: function(form, data, transactionId) {
    fetch('/api/submit-lead', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ...data, transaction_id: transactionId })
    })
    .then(response => {
      if (response.ok) {
        // Show success message or redirect
        form.style.display = 'none';
        document.getElementById('form-success').style.display = 'block';
      }
    })
    .catch(error => {
      console.error('Form submission error:', error);
    });
  }
};

document.addEventListener('DOMContentLoaded', () => FormTracker.init());
