📦 Data Layer Implementation

Critical Requirement

The Data Layer MUST be initialized BEFORE the GTM container code loads. Place this script at the very top of your <head> section.

data-layer-init.js — Place BEFORE GTM script in <head>
// ============================================
// ComCreate B2B Landing Page - Data Layer Initialization
// Must load BEFORE Google Tag Manager
// ============================================

window.dataLayer = window.dataLayer || [];

// Helper: Get URL parameter value
function getUrlParam(name) {
  return new URLSearchParams(window.location.search).get(name);
}

// Helper: Get or create session ID
function getSessionId() {
  let sessionId = sessionStorage.getItem('cc_session_id');
  if (!sessionId) {
    sessionId = 'session_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
    sessionStorage.setItem('cc_session_id', sessionId);
  }
  return sessionId;
}

// Store GCLID in cookie for cross-session attribution (90 days)
(function() {
  const gclid = getUrlParam('gclid');
  if (gclid) {
    const expiry = new Date();
    expiry.setDate(expiry.getDate() + 90);
    document.cookie = 'gclid=' + gclid + ';expires=' + expiry.toUTCString() + ';path=/;SameSite=Lax';
  }
})();

// Store all UTM parameters in sessionStorage
(function() {
  const utmParams = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_content', 'utm_term'];
  utmParams.forEach(param => {
    const value = getUrlParam(param);
    if (value) {
      sessionStorage.setItem(param, value);
    }
  });
})();

// Get stored or current UTM value
function getUtm(param) {
  return getUrlParam(param) || sessionStorage.getItem(param) || 'direct';
}

// Push initial page data when DOM is ready
dataLayer.push({
  'event': 'page_data_ready',
  'page': {
    'type': 'b2b_sales_landing',
    'path': window.location.pathname,
    'title': document.title,
    'referrer': document.referrer
  },
  'session': {
    'id': getSessionId(),
    'landing_page': sessionStorage.getItem('landing_page') || window.location.href
  },
  'campaign': {
    'source': getUtm('utm_source'),
    'medium': getUtm('utm_medium'),
    'campaign': getUtm('utm_campaign'),
    'content': getUtm('utm_content'),
    'term': getUtm('utm_term'),
    'gclid': getUrlParam('gclid') || document.cookie.match(/gclid=([^;]+)/)?.[1] || 'none'
  }
});

// Store landing page on first visit
if (!sessionStorage.getItem('landing_page')) {
  sessionStorage.setItem('landing_page', window.location.href);
}

// Track page load time for engagement metrics
window.pageLoadTime = Date.now();

console.log('[ComCreate] Data layer initialized');

Lead Value Configuration

lead-value-config.js
// ============================================
// Dynamic Lead Value Calculator
// Based on package selection and business type
// ============================================

const ComCreateLeadValues = {
  // Base values by package tier
  packages: {
    'starter': 75,    // $749/mo × 10% close rate
    'growth': 110,    // $1,099/mo × 10% close rate (most common)
    'scale': 150      // $1,499/mo × 10% close rate
  },

  // Multipliers by business type (based on average deal size)
  businessMultipliers: {
    'plumber': 1.0,
    'hvac': 1.2,        // Higher ticket services
    'electrician': 1.0,
    'roofer': 1.3,      // High ticket, longer cycles
    'dentist': 1.5,     // High LTV
    'lawyer': 1.4,      // High LTV
    'other': 1.0
  },

  // Discovery call booked is worth more (higher intent)
  discoveryCallBonus: 1.5
};

/**
 * Calculate lead value based on package and business type
 * @param {string} packageTier - 'starter', 'growth', or 'scale'
 * @param {string} businessType - Type of business
 * @param {boolean} isDiscoveryCall - Whether they booked a call
 * @returns {number} Lead value in dollars
 */
function calculateLeadValue(packageTier, businessType, isDiscoveryCall = false) {
  const baseValue = ComCreateLeadValues.packages[packageTier] || 100;
  const multiplier = ComCreateLeadValues.businessMultipliers[businessType] || 1.0;
  const callBonus = isDiscoveryCall ? ComCreateLeadValues.discoveryCallBonus : 1.0;

  return Math.round(baseValue * multiplier * callBonus);
}

/**
 * Generate unique transaction ID for deduplication
 * @returns {string} Unique transaction ID
 */
function generateTransactionId() {
  return 'cc_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
}

📝 Multi-Step Form Tracking

Why Multi-Step?

Multi-step forms convert 743% better than single-step forms. We track each step to identify where prospects drop off and optimize accordingly.

Form Structure

Step Fields Event to Fire Purpose
Step 1 Business type, Location form_start Low friction entry, qualify lead type
Step 2 Package selection, Monthly budget package_selected Intent signal, qualification
Step 3 Name, Email, Phone quote_request Capture lead info
multi-step-form-tracking.js
// ============================================
// ComCreate Multi-Step Form Tracking
// Tracks progression through the lead capture form
// ============================================

const FormTracker = {
  formData: {},
  currentStep: 0,
  formStarted: false,

  init: function() {
    const form = document.querySelector('#lead-capture-form');
    if (!form) {
      console.warn('[FormTracker] Form not found');
      return;
    }

    this.form = form;
    this.bindEvents();
    console.log('[FormTracker] Initialized');
  },

  bindEvents: function() {
    // Track form start (first interaction)
    this.form.querySelectorAll('input, select').forEach(field => {
      field.addEventListener('focus', () => this.trackFormStart(), { once: true });
    });

    // Track step progression
    this.form.querySelectorAll('[data-step-trigger]').forEach(btn => {
      btn.addEventListener('click', (e) => this.trackStepCompletion(e));
    });

    // Track final submission
    this.form.addEventListener('submit', (e) => this.trackSubmission(e));
  },

  trackFormStart: function() {
    if (this.formStarted) return;
    this.formStarted = true;

    dataLayer.push({
      'event': 'form_start',
      'form_id': 'lead_capture',
      'form_step': 1,
      'time_to_engage': Math.round((Date.now() - window.pageLoadTime) / 1000)
    });

    console.log('[FormTracker] Form started');
  },

  trackStepCompletion: function(e) {
    const trigger = e.target.closest('[data-step-trigger]');
    const nextStep = parseInt(trigger.dataset.stepTrigger);

    // Collect data from current step
    this.collectStepData(this.currentStep);

    // Step 2 completion = package selected
    if (nextStep === 3 && this.formData.package_selected) {
      const leadValue = calculateLeadValue(
        this.formData.package_selected,
        this.formData.business_type,
        false
      );

      dataLayer.push({
        'event': 'package_selected',
        'form_step': 2,
        'package_selected': this.formData.package_selected,
        'business_type': this.formData.business_type,
        'lead_value': leadValue
      });

      console.log('[FormTracker] Package selected:', this.formData.package_selected);
    }

    this.currentStep = nextStep;
  },

  collectStepData: function(step) {
    const stepContainer = this.form.querySelector(`[data-step="${step}"]`);
    if (!stepContainer) return;

    stepContainer.querySelectorAll('input, select').forEach(field => {
      if (field.name && field.value) {
        this.formData[field.name] = field.value;
      }
    });
  },

  trackSubmission: function(e) {
    e.preventDefault();

    // Collect final step data
    this.collectStepData(3);

    const transactionId = generateTransactionId();
    const leadValue = calculateLeadValue(
      this.formData.package_selected || 'growth',
      this.formData.business_type || 'other',
      false
    );

    // Push conversion event
    dataLayer.push({
      'event': 'quote_request',
      'form_id': 'lead_capture',
      'transaction_id': transactionId,
      'lead_value': leadValue,
      'package_selected': this.formData.package_selected,
      'business_type': this.formData.business_type,
      'user_data': {
        'email': this.normalizeEmail(this.formData.email),
        'phone_number': this.normalizePhone(this.formData.phone),
        'business_name': this.formData.business_name
      }
    });

    console.log('[FormTracker] Quote request submitted', { transactionId, leadValue });

    // Submit to backend
    this.submitToBackend(transactionId);
  },

  normalizeEmail: function(email) {
    return email ? email.toLowerCase().trim() : '';
  },

  normalizePhone: function(phone) {
    if (!phone) return '';
    const cleaned = phone.replace(/\D/g, '');
    return cleaned.length === 10 ? '+1' + cleaned : '+' + cleaned;
  },

  submitToBackend: function(transactionId) {
    const payload = {
      ...this.formData,
      transaction_id: transactionId,
      utm_source: sessionStorage.getItem('utm_source') || 'direct',
      utm_medium: sessionStorage.getItem('utm_medium') || 'none',
      utm_campaign: sessionStorage.getItem('utm_campaign') || 'none',
      gclid: document.cookie.match(/gclid=([^;]+)/)?.[1] || 'none',
      landing_page: sessionStorage.getItem('landing_page'),
      submitted_at: new Date().toISOString()
    };

    fetch('/api/leads', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    })
    .then(response => {
      if (response.ok) {
        // Show success state
        this.form.style.display = 'none';
        document.querySelector('#form-success')?.classList.add('visible');

        // Optionally redirect to calendar booking
        // window.location.href = '/book-call?txn=' + transactionId;
      }
    })
    .catch(error => {
      console.error('[FormTracker] Submission error:', error);
    });
  }
};

// Initialize when DOM is ready
document.addEventListener('DOMContentLoaded', () => FormTracker.init());

📅 Cal.com Booking Integration

Why Calendar Integration Matters

66.7% of leads who book a discovery call convert to customers. Tracking calendar bookings is critical for measuring true conversion intent and optimizing Google Ads bidding.

Cal.com Embed Tracking

cal-booking-tracking.js
// ============================================
// Cal.com Booking Event Tracking
// Listens for Cal.com postMessage events
// ============================================

(function() {
  // Store form data from previous steps for attribution
  let preBookingData = {};

  // Capture pre-booking data from sessionStorage
  function getPreBookingData() {
    return {
      package_selected: sessionStorage.getItem('cc_package_selected') || 'growth',
      business_type: sessionStorage.getItem('cc_business_type') || 'other',
      utm_source: sessionStorage.getItem('utm_source') || 'direct',
      utm_campaign: sessionStorage.getItem('utm_campaign') || 'none',
      gclid: document.cookie.match(/gclid=([^;]+)/)?.[1] || 'none'
    };
  }

  // Listen for Cal.com booking events
  window.addEventListener('message', function(event) {
    // Verify origin (adjust domain as needed)
    if (!event.origin.includes('cal.com')) return;

    const data = event.data;

    // Cal.com sends different event types
    if (data.event === 'booking:confirmed' || data.type === 'booking_successful') {
      handleBookingConfirmed(data);
    }

    if (data.event === 'booking:created') {
      handleBookingConfirmed(data);
    }
  });

  function handleBookingConfirmed(calData) {
    const preData = getPreBookingData();
    const transactionId = generateTransactionId();

    const leadValue = calculateLeadValue(
      preData.package_selected,
      preData.business_type,
      true // Discovery call bonus
    );

    // Push conversion event
    dataLayer.push({
      'event': 'discovery_call_booked',
      'transaction_id': transactionId,
      'lead_value': leadValue,
      'package_selected': preData.package_selected,
      'business_type': preData.business_type,
      'booking_data': {
        'event_type': calData.eventType || 'discovery-call',
        'start_time': calData.startTime || null,
        'attendee_email': calData.attendeeEmail || null
      },
      'user_data': {
        'email': calData.attendeeEmail || null,
        'first_name': calData.attendeeName?.split(' ')[0] || null
      }
    });

    console.log('[CalTracking] Discovery call booked', {
      transactionId,
      leadValue,
      package: preData.package_selected
    });
  }
})();

🔐 Enhanced Conversions Setup

Why Enhanced Conversions?

Enhanced Conversions improve conversion measurement accuracy by 5-15% by sending hashed (SHA-256) user data to Google. This is especially important for B2B where conversion paths are longer.

2

2. Select "Code" or "Manual configuration"
3. Map the data layer variables

Privacy Note

Google automatically hashes user data using SHA-256 before transmission. The raw data never leaves the user's browser. This complies with privacy regulations when properly disclosed in your privacy policy.
